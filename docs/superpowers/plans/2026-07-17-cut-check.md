```ts
import type { CutCheckKind, CutCheckSeverity } from "./record.ts";

export interface CutCheckSegment {
  id: string;
  start_seconds: number;
  end_seconds: number;
  text: string;
}

export interface EngineFinding {
  kind: CutCheckKind;
  severity: CutCheckSeverity;
  title: string;
  detail: string;
  segment_ids: string[];
}

const HOOK_WINDOW_SECONDS = 3;
const LATE_HOOK_WINDOW_SECONDS = 8;
const DEAD_AIR_SECONDS = 1.5;
const RUSHED_WORDS_PER_SECOND = 2.5;
const CTA_WINDOW_SECONDS = 5;
const MAX_FINDINGS_PER_KIND = 3;

const QUESTION = /\?/;
const NUMERAL =
  /(^\s*\d|\$[\d,.]+\s*(million|billion|thousand|k)?|\d+(\.\d+)?\s*(%|percent|million|billion|x)\b)/i;
const AUTHORITY =
  /\b(only|best|worst|never|everyone|nobody|secret|truth|proven|future|staggering|blueprint)\b/i;
const DIRECT_ADDRESS = /\byou(r)?\b/i;
const PIVOT =
  /\b(but|so|and it gets better|here's the thing|the catch|the problem|not just)\b/i;
const PROOF_TOKEN =
  /(\$[\d,.]+\s*(million|billion|thousand|k)?|\d+(\.\d+)?\s*(%|percent)|\d[\d,.]*\s*(gallons|cars|facilities|years|hours|days)\b)/gi;
const CTA =
  /\b(follow|subscribe|learn more|link in bio|comment|share|visit|call|contact|book|get started)\b/i;

function wordsOf(text: string): number {
  return text.trim().split(/\s+/).filter(Boolean).length;
}

function isHooky(text: string): boolean {
  return QUESTION.test(text) || NUMERAL.test(text) || AUTHORITY.test(text) || DIRECT_ADDRESS.test(text);
}

function proofTokensOf(text: string): string[] {
  return (text.match(PROOF_TOKEN) ?? []).map((token) => token.replace(/\s+/g, " ").trim());
}

/** Normalize a proof token to its numeric core for script-vs-cut comparison. */
function numericCore(token: string): string {
  const match = token.replace(/,/g, "").match(/[\d.]+/);
  return match ? match[0] : token;
}

export function runDeterministicCutCheck(input: {
  segments: CutCheckSegment[];
  scriptRef?: string | null;
  durationSeconds?: number | null;
}): EngineFinding[] {
  const segments = [...input.segments].sort((a, b) => a.start_seconds - b.start_seconds);
  if (segments.length === 0) {
    return [
      {
        kind: "hook",
        severity: "info",
        title: "Transcript required",
        detail:
          "Cut Check needs a transcript for this version before it can reason about hook, pacing, structure, proof, or CTA. Process the transcript, then re-run.",
        segment_ids: [],
      },
    ];
  }

  const findings: EngineFinding[] = [];
  const duration =
    input.durationSeconds ?? Math.max(...segments.map((segment) => segment.end_seconds));
  const fullText = segments.map((segment) => segment.text).join(" ");

  // hook
  const opening = segments.filter((segment) => segment.start_seconds < HOOK_WINDOW_SECONDS);
  const late = segments.filter(
    (segment) =>
      segment.start_seconds >= HOOK_WINDOW_SECONDS &&
      segment.start_seconds < LATE_HOOK_WINDOW_SECONDS,
  );
  if (opening.some((segment) => isHooky(segment.text))) {
    findings.push({
      kind: "hook",
      severity: "info",
      title: "Hook lands in the opening",
      detail: "A hook pattern (question, stat, authority claim, or direct address) lands inside the first 3 seconds.",
      segment_ids: opening.filter((segment) => isHooky(segment.text)).map((segment) => segment.id),
    });
  } else if (late.some((segment) => isHooky(segment.text))) {
    findings.push({
      kind: "hook",
      severity: "attention",
      title: "Hook lands late",
      detail: `The first hook pattern appears after the 3-second mark. Short-form openings are decided in under 3 seconds — move the hook up.`,
      segment_ids: late.filter((segment) => isHooky(segment.text)).map((segment) => segment.id),
    });
  } else {
    findings.push({
      kind: "hook",
      severity: "critical",
      title: "No hook pattern in the opening",
      detail:
        "No question, statistic, authority claim, or direct address appears in the first 8 seconds. The cut opens cold.",
      segment_ids: opening.map((segment) => segment.id),
    });
  }

  // pacing — dead air + rushed delivery (capped)
  for (let index = 0; index < segments.length - 1 && findings.filter((f) => f.kind === "pacing").length < MAX_FINDINGS_PER_KIND; index += 1) {
    const current = segments[index];
    const next = segments[index + 1];
    const gap = next.start_seconds - current.end_seconds;
    if (gap > DEAD_AIR_SECONDS) {
      findings.push({
        kind: "pacing",
        severity: "attention",
        title: "Dead air between beats",
        detail: `A ${gap.toFixed(1)}s gap between "${current.text.slice(0, 40)}…" and the next beat. Tighten or cover with B-roll.`,
        segment_ids: [current.id, next.id],
      });
    }
  }
  let rushedRun: CutCheckSegment[] = [];
  const rushed: CutCheckSegment[] = [];
  for (const segment of segments) {
    const wps = wordsOf(segment.text) / Math.max(segment.end_seconds - segment.start_seconds, 0.1);
    if (wps > RUSHED_WORDS_PER_SECOND) {
      rushedRun.push(segment);
    } else {
      if (rushedRun.length >= 3) rushed.push(...rushedRun);
      rushedRun = [];
    }
  }
  if (rushedRun.length >= 3) rushed.push(...rushedRun);
  if (rushed.length > 0) {
    findings.push({
      kind: "pacing",
      severity: "attention",
      title: "Delivery runs rushed",
      detail: `${rushed.length} consecutive segments exceed ${RUSHED_WORDS_PER_SECOND} words/second. Give key lines room to land.`,
      segment_ids: rushed.map((segment) => segment.id),
    });
  }
  if (!findings.some((f) => f.kind === "pacing")) {
    findings.push({
      kind: "pacing",
      severity: "info",
      title: "Pacing holds",
      detail: "No dead air over 1.5s and no rushed runs detected.",
      segment_ids: [],
    });
  }

  // structure
  const pivots = segments.filter((segment) => PIVOT.test(segment.text));
  findings.push(
    pivots.length > 0
      ? {
          kind: "structure",
          severity: "info",
          title: "Structural pivots present",
          detail: `${pivots.length} curiosity-loop / pivot marker${pivots.length === 1 ? "" : "s"} keep the script moving.`,
          segment_ids: pivots.map((segment) => segment.id),
        }
      : {
          kind: "structure",
          severity: "attention",
          title: "No structural pivot detected",
          detail:
            'No pivot markers ("but", "so", "and it gets better"…) — the cut reads as one flat run. Add a turn.',
          segment_ids: [],
        },
  );

  // proof
  const cutProofs = proofTokensOf(fullText);
  const scriptProofs = input.scriptRef ? proofTokensOf(input.scriptRef) : [];
  if (input.scriptRef && scriptProofs.length > 0) {
    const cutCores = new Set(cutProofs.map(numericCore));
    const missing = scriptProofs.filter((token) => !cutCores.has(numericCore(token)));
    for (const token of missing.slice(0, MAX_FINDINGS_PER_KIND)) {
      findings.push({
        kind: "proof",
        severity: "critical",
        title: "Script proof point missing from cut",
        detail: `The script's proof point "${token}" does not appear in the cut's transcript.`,
        segment_ids: [],
      });
    }
    if (missing.length === 0) {
      findings.push({
        kind: "proof",
        severity: "info",
        title: "Proof stack intact",
        detail: `All ${scriptProofs.length} script proof point${scriptProofs.length === 1 ? "" : "s"} appear in the cut.`,
        segment_ids: segments.filter((segment) => PROOF_TOKEN.test(segment.text)).map((segment) => segment.id),
      });
    }
  } else {
    findings.push(
      cutProofs.length > 0
        ? {
            kind: "proof",
            severity: "info",
            title: "Quantified proof present",
            detail: `${cutProofs.length} quantified proof point${cutProofs.length === 1 ? "" : "s"} detected.`,
            segment_ids: segments.filter((segment) => PROOF_TOKEN.test(segment.text)).map((segment) => segment.id),
          }
        : {
            kind: "proof",
            severity: "attention",
            title: "No quantified proof points",
            detail: "No numbers, percentages, or measured claims detected. Proof stacks carry credibility.",
            segment_ids: [],
          },
    );
  }

  // cta
  const closing = segments.filter((segment) => segment.end_seconds >= duration - CTA_WINDOW_SECONDS);
  const closingText = closing.map((segment) => segment.text).join(" ");
  findings.push(
    CTA.test(closingText)
      ? {
          kind: "cta",
          severity: "info",
          title: "Call to action present",
          detail: "A CTA lands in the closing window.",
          segment_ids: closing.map((segment) => segment.id),
        }
      : {
          kind: "cta",
          severity: "attention",
          title: "No call to action in the close",
          detail: "The final 5 seconds carry no CTA. Tell the viewer what to do next.",
          segment_ids: closing.map((segment) => segment.id),
        },
  );

  return findings;
}
```

```ts
import { getProjectAccess } from "@/lib/access-control";
import { requireAuth } from "@/lib/auth";
import { runDeterministicCutCheck } from "@/lib/covideopro/cut-check";
import { getSupabase } from "@/lib/supabase";

const UUID_PATTERN = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

function errorResponse(error: string, status: number) {
  return NextResponse.json({ error }, { status });
}

export async function POST(req: Request) {
  const user = await requireAuth();
  if (!user) return errorResponse("Unauthorized", 401);

  let body: unknown;
  try {
    body = await req.json();
  } catch {
    return errorResponse("Request body must be a JSON object", 400);
  }
  const { project_id: projectId, asset_id: assetId, version_id: versionId, script_ref: scriptRef } =
    (body ?? {}) as Record<string, unknown>;
  if (
    typeof projectId !== "string" || !UUID_PATTERN.test(projectId) ||
    typeof assetId !== "string" || !UUID_PATTERN.test(assetId) ||
    typeof versionId !== "string" || !UUID_PATTERN.test(versionId)
  ) {
    return errorResponse("project_id, asset_id, and version_id must be UUIDs", 400);
  }
  if (scriptRef !== undefined && scriptRef !== null && typeof scriptRef !== "string") {
    return errorResponse("script_ref must be a string", 400);
  }

  const access = await getProjectAccess(projectId, user.id, "editor");
  if (!access.ok) return errorResponse(access.error, access.status);

  const supabase = getSupabase();

  // Idempotent: one active run per (version, engine).
  const { data: active } = await supabase
    .from("cut_checks")
    .select("id, status")
    .eq("version_id", versionId)
    .eq("engine", "deterministic-v1")
    .in("status", ["queued", "running"])
    .maybeSingle();
  if (active) return NextResponse.json({ check_id: active.id, status: active.status }, { status: 200 });

  // Verify the version belongs to this project before any write.
  const { data: version } = await supabase
    .from("versions")
    .select("id, asset_id")
    .eq("id", versionId)
    .eq("asset_id", assetId)
    .maybeSingle();
  if (!version) return errorResponse("Version not found for asset", 404);

  const { data: transcription } = await supabase
    .from("transcriptions")
    .select("segments")
    .eq("version_id", versionId)
    .eq("asset_id", assetId)
    .eq("status", "completed")
    .maybeSingle();

  const rawSegments = Array.isArray(transcription?.segments) ? transcription.segments : [];
  const segments = rawSegments
    .filter((s): s is { start: number; end: number; text: string } =>
      typeof s?.start === "number" && typeof s?.end === "number" && typeof s?.text === "string")
    .map((s, index) => ({ id: `seg-${index}`, start_seconds: s.start, end_seconds: s.end, text: s.text }));

  const engineFindings = runDeterministicCutCheck({
    segments,
    scriptRef: typeof scriptRef === "string" ? scriptRef : null,
  });

  const { data: check, error: insertError } = await supabase
    .from("cut_checks")
    .insert({
      project_id: projectId,
      asset_id: assetId,
      version_id: versionId,
      script_ref: typeof scriptRef === "string" ? scriptRef : null,
      engine: "deterministic-v1",
      status: "completed",
      created_by: user.id,
    })
    .select("id")
    .single();
  if (insertError || !check) return errorResponse("Failed to persist cut check", 500);

  const findingRows = engineFindings.map((finding) => ({
    check_id: check.id,
    kind: finding.kind,
    severity: finding.severity,
    title: finding.title,
    detail: finding.detail,
    segment_ids: finding.segment_ids,
    record_citations: [{ type: "version", id: versionId }],
    created_by: user.id,
  }));
  const { error: findingsError } = await supabase.from("cut_check_findings").insert(findingRows);
  if (findingsError) return errorResponse("Failed to persist findings", 500);

  await supabase.from("activity_log").insert({
    project_id: projectId,
    actor_id: user.id,
    action: "cut_check_completed",
    details: { findings: String(findingRows.length) },
  });

  return NextResponse.json({ check_id: check.id, status: "completed" }, { status: 201 });
}
```

```tsx
const CUT_CHECK_SEVERITY_CLASS: Record<string, string> = {
  info: "status-active",
  attention: "status-pending",
  critical: "status-version",
};

export function CutCheckSection({ projectId, demoMode, onNotice }: SectionProps) {
  const workspace = useDemoWorkspace();
  const checks = useMemo(
    () => workspace.cutChecks.filter((check) => check.project_id === projectId),
    [workspace.cutChecks, projectId],
  );
  const findingsByCheck = useMemo(() => {
    const map = new Map<string, typeof workspace.cutCheckFindings>();
    for (const finding of workspace.cutCheckFindings) {
      const list = map.get(finding.check_id) ?? [];
      list.push(finding);
      map.set(finding.check_id, list);
    }
    return map;
  }, [workspace.cutCheckFindings]);

  if (!demoMode) {
    return <SectionEmpty title="Cut Check" body="Cut Check runs in the local workspace during demo review." />;
  }

  const assets = workspace.assets.filter((asset) => asset.project_id === projectId);

  return (
    <section aria-label="Cut Check">
      <header style={{ display: "flex", alignItems: "center", justifyContent: "space-between" }}>
        <div>
          <h2>Cut Check</h2>
          <p className="empty-state-text">
            Advisory AI findings on a cut before it goes out. AI findings are advisory; the editor decides.
          </p>
        </div>
        <button
          type="button"
          onClick={() => {
            const asset = assets[0];
            if (!asset) {
              onNotice("Upload media before running Cut Check.");
              return;
            }
            const result = runCutCheck({
              projectId,
              assetId: asset.id,
              versionId: asset.current_version_id ?? "v1",
            });
            onNotice(result.ok ? "Cut Check completed." : result.reason);
          }}
        >
          Run Cut Check
        </button>
      </header>

      {checks.length === 0 ? (
        <SectionEmpty title="No Cut Checks yet" body="Run one from a version to get hook, pacing, structure, proof, and CTA findings." />
      ) : (
        checks.map((check) => (
          <article key={check.id} className="cockpit-record-card" style={{ marginTop: 16 }}>
            <header style={{ display: "flex", gap: 8, alignItems: "center" }}>
              <ScanSearch size={16} />
              <strong>{new Date(check.created_at).toLocaleString()}</strong>
              <span className="demo-pill">{check.engine}</span>
              <span className={check.status === "completed" ? "status-active" : "status-pending"}>{check.status}</span>
            </header>
            <ul style={{ listStyle: "none", padding: 0, marginTop: 12, display: "grid", gap: 8 }}>
              {(findingsByCheck.get(check.id) ?? []).map((finding) => (
                <li key={finding.id} className="cockpit-record-card" style={{ padding: 12 }}>
                  <div style={{ display: "flex", gap: 8, alignItems: "center" }}>
                    <span className={CUT_CHECK_SEVERITY_CLASS[finding.severity]}>{finding.severity}</span>
                    <span className="demo-pill">{finding.kind}</span>
                    <strong>{finding.title}</strong>
                    <span style={{ marginLeft: "auto" }} className={finding.status === "pending" ? "status-pending" : "status-active"}>
                      {finding.status}
                    </span>
                  </div>
                  <p className="empty-state-text" style={{ marginTop: 6 }}>{finding.detail}</p>
                  {finding.status === "pending" ? (
                    <div style={{ display: "flex", gap: 8, marginTop: 8 }}>
                      <button type="button" onClick={() => {
                        const result = resolveCutCheckFinding({ checkId: check.id, findingId: finding.id, action: "accepted" });
                        onNotice(result.ok ? "Finding accepted." : result.reason);
                      }}>Accept</button>
                      <button type="button" onClick={() => {
                        const result = resolveCutCheckFinding({ checkId: check.id, findingId: finding.id, action: "dismissed" });
                        onNotice(result.ok ? "Finding dismissed." : result.reason);
                      }}>Dismiss</button>
                    </div>
                  ) : null}
                </li>
              ))}
            </ul>
          </article>
        ))
      )}
    </section>
  );
}
```
